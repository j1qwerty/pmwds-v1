# ============================================================================
# common.txt - elements shared by every page
#
# Shared modal chrome, toasts, nav, and control patterns. Page maps reference
# these group ids instead of repeating the selectors.
#
# Selector attributes the driver understands, in priority order:
#   css=      raw selector, wins over everything
#   testid=   data-testid
#   aria=     aria-label
#   title=    title attribute (most icon-only buttons use only this)
#   placeholder=
#   role= + name=
#   label=
#   text=
#   root=     scope the search, e.g. root=.modal
#   nth=      comma list of indexes to try, e.g. nth=0,1
#   hover=1   hover the row first (for group-hover-only buttons)
# ============================================================================
